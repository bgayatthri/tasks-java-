class Solution {
    public List<String> findAndReplacePattern(String[] words, String pattern) {
         List<String> res = new ArrayList<>();

        for (String w : words) {
            int[] m1 = new int[26];
            int[] m2 = new int[26];
            boolean x = true;

            for (int i = 0; i < w.length(); i++) {
                int a = w.charAt(i) - 'a';
                int b = pattern.charAt(i) - 'a';

                if (m1[a] != m2[b]) {
                    x = false;
                    break;
                }

                m1[a] = m2[b] = i + 1;
            }

            if (x) res.add(w);
        }
        return res;
    }
}
